// URLs de extracción en tiempo real usando un proxy CORS libre const PROXY = "https://allorigins.win"; const TARGET = encodeURIComponent("https://lottery.jackpocket.com/en/lottery-results/puerto-rico"); // Al arrancar el "Sistema Operativo Win98" window.onload = function() { fetchLottoRealTime(); renderHistory(); }; async function fetchLottoRealTime() { // Intentar buscar un contenedor de texto de estado en la UI, o usar la consola si no existe console.log("Win98_Network: Estableciendo conexión dial-up con Jackpocket..."); try { const response = await fetch(`${PROXY}${TARGET}`); if (!response.ok) throw new Error("Error de red"); const resData = await response.json(); // Convertir el texto HTML plano en un documento DOM para buscar elementos const parser = new DOMParser(); const doc = parser.parseFromString(resData.contents, "text/html"); // Objeto temporal para mapear lo que encontremos en la web window.winningMap = {}; // Extraer los bloques de juegos de la página de Jackpocket const gameRows = doc.querySelectorAll('.v-list-item'); gameRows.forEach(row => { const titleEl = row.querySelector('.v-list-item__title'); const subtitleEl = row.querySelector('.v-list-item__subtitle'); if (titleEl && subtitleEl) { const title = titleEl.textContent.trim().toUpperCase(); // Ej: "PEGA 3 NOCHE" const rawNumbers = subtitleEl.textContent.trim(); // Ej: "9 2 9 Wild Ball: 5" // Limpiar y extraer solo los dígitos principales para nuestro buscador rápido const digitsOnly = rawNumbers.replace(/Wild Ball:.*\$/i, '').replace(/\D/g, ''); if (digitsOnly) { window.winningMap[digitsOnly] = `¡Ganador en ${title}!`; } // Actualizar visualmente las tarjetas de la interfaz Win98 de manera dinámica actualizarInterfazGrafica(title, rawNumbers); } }); console.log("Sincronización Win98 completada con éxito. Datos cargados."); } catch (error) { console.error("Fallo de red Win98: Usando contingencia local por desconexión."); cargarBackupLocal(); } } // Busca las tarjetas de juego en el HTML y actualiza sus esferas amarillas function actualizarInterfazGrafica(titulo, textoNumeros) { // Extrae la parte antes de la Wild Ball const partePrincipal = textoNumeros.replace(/Wild Ball:.*\$/i, '').trim(); const numerosArray = partePrincipal.split(/\s+/).filter(n => n.length > 0); // Extrae la Wild Ball si existe const wbMatch = textoNumeros.match(/Wild Ball:\s*(\d+)/i); const wbValue = wbMatch ? wbMatch[1] : null; // Mapeo selectivo por títulos comunes de la lotería de Puerto Rico let targetCardIndex = -1; if (titulo.includes("PEGA 2")) targetCardIndex = 0; else if (titulo.includes("PEGA 3")) targetCardIndex = 1; else if (titulo.includes("PEGA 4")) targetCardIndex = 2; else if (titulo.includes("LOTO") || titulo.includes("POWERBALL")) targetCardIndex = 3; if (targetCardIndex !== -1) { const card = document.querySelectorAll('.game-card')[targetCardIndex]; if (card) { const titleSpan = card.querySelector('.game-header span:first-child'); if (titleSpan) titleSpan.textContent = titulo; // Actualiza el nombre exacto del sorteo (Día/Noche) const numbersRow = card.querySelector('.numbers-row'); if (numbersRow) { let htmlBolas = numerosArray.map(n => `
${n}
`).join(''); if (wbValue) { htmlBolas += `WB:
${wbValue}
`; } numbersRow.innerHTML = htmlBolas; } } } } // Sistema de respaldo por si el proxy falla o se satura temporalmente function cargarBackupLocal() { window.winningMap = { "64": "¡Ganador simulado en Pega 2!", "611": "¡Ganador simulado en Pega 3!", "4108": "¡Ganador simulado en Pega 4!" }; } // Lógica del Verificador de números vinculada al LocalStorage function checkNumber() { const val = document.getElementById('userNum').value.trim(); const msg = document.getElementById('resultMsg'); if (!window.winningMap) cargarBackupLocal(); if (!val) { msg.textContent = "Error: Input vacío."; msg.style.color = "red"; return; } if (window.winningMap[val]) { msg.textContent = `[OK] ${window.winningMap[val]}`; msg.style.color = "green"; saveToHistory(val, "Ganador"); } else { msg.textContent = `[X] El número ${val} no registra premios mayores en este sorteo.`; msg.style.color = "var(--win-blue)"; saveToHistory(val, "No Premiado"); } } function saveToHistory(num, status) { let history = JSON.parse(localStorage.getItem('win98_lotto_history')) || []; history.unshift({ num, status, time: new Date().toLocaleTimeString() }); if(history.length > 5) history.pop(); localStorage.setItem('win98_lotto_history', JSON.stringify(history)); renderHistory(); } function renderHistory() { console.log("Historial de consultas del registro de Windows sincronizado."); }
